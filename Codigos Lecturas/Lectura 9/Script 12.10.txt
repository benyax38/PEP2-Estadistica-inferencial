library(dplyr)
library(ggpubr)
library(glue)
library(purrr)
library(rsample)
library(tibble)
library(tidyr)

# Crear muestras iniciales
notas_I <- c(5.4, 4.7, 6.3, 2.9, 5.9, 5.1, 2.1, 6.2, 1.6, 6.7, 3.0, 3.3,
            5.0, 4.1, 3.3, 3.4, 1.2, 3.8, 5.8, 4.2)
notas_U <- c(4.0, 4.1, 4.3, 4.3, 4.3, 4.2, 4.3, 4.3, 4.4, 4.1, 4.3, 4.0)

# Crear matriz de datos con las muestras independientes
Muestra <- factor(c(rep("I", length(notas_I)), rep("U", length(notas_U))))
datos_largos <- tibble(Muestra, Nota = c(notas_I, notas_U))

# Crear las remuestras
R <- 5999
set.seed(432)
remuestras <- permutations(datos_largos, permute = Nota, times = R, apparent = TRUE)

# Función que devuelve los estadísticos de interés para la diferencia de dos varianzas
# en una remuestra.
dif_dos_varianzas <- function(remuestra, ...) {
    analysis(remuestra) |> group_by(Muestra) |> summarise(v = var(Nota)) |>
        summarise(v[1] - v[2]) |> pull()
}

# Obtener la distribución permutada de la diferencia de las medias
dist_perm <- remuestras |> mutate(dif_v = map(splits, dif_dos_varianzas)) |> unnest(dif_v)
