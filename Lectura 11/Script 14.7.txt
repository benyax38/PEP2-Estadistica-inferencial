library(broom)
library(dplyr)
library(formatR)
library(scales)

# Cargar y filtrar los datos
datos <- mtcars |> filter(wt > 2 & wt < 5) |>
    mutate(across(c("cyl", "vs", "am", "gear", "carb"), as.factor))

# Ajustar el modelo nulo y el modelo completo
nulo <- lm(hp ~ 1, data = datos)
completo <- lm(hp ~ ., data = datos)

##### Selección hacia adelante #####

# Evaluar las variables para seleccionar el primer predictor
paso <- add1(nulo, scope = completo, test = "F") |> tidy() |>
    mutate(p.value = format(pvalue(p.value), justify = "right")) |> arrange(AIC)
cat("", paste("Modelo:", format(formula(nulo))), format(paso), sep = "\n")

# Agregar la variable que logra la mayor reducción en AIC
modelo <- update(nulo, . ~ . + cyl)

# Evaluar las variables para seleccionar el segundo predictor
paso <- add1(modelo, scope = completo, test = "F") |> tidy() |>
    mutate(p.value = format(pvalue(p.value), justify = "right")) |> arrange(AIC)
cat("", paste("Modelo:", format(formula(modelo))), format(paso), sep = "\n")

# Agregar la variable que logra la mayor reducción en AIC
modelo <- update(modelo, . ~ . + carb)

# Mostrar los coeficientes del modelo conseguido
cat("\nCoeficientes para los predictores del modelo obtenido:\n")
print(round(modelo[["coefficients"]], 3), width = 56)

##### Selección hacia atrás #####

# Evaluar la eliminación de uno de los predictores del modelo
paso <- drop1(completo, test = "F") |> tidy() |>
    mutate(p.value = format(pvalue(p.value), justify = "right")) |> arrange(desc(statistic))
fmla_str <- tidy_source(text = format(formula(completo)), width.cutoff = 47, output = FALSE)
cat("", paste("Modelo:", fmla_str[[1]]), format(paso), sep = "\n")

# Quitar el predictor que menos aporta (con menor estadístico F)
modelo <- update(completo, . ~ . - wt)

# Evaluar la eliminación de uno de los predictores del modelo
paso <- drop1(modelo, test = "F") |> tidy() |>
    mutate(p.value = format(pvalue(p.value), justify = "right")) |> arrange(desc(statistic))
fmla_str <- tidy_source(text = format(formula(modelo)), width.cutoff = 47, output = FALSE)
cat("", paste("Modelo:", fmla_str[[1]]), format(paso), sep = "\n")

# Quitar el predictor que menos aporta (con menor estadístico F)
modelo <- update(modelo, . ~ . - drat)

# Mostrar los coeficientes del modelo conseguido
cat("\nCoeficientes para los predictores del modelo obtenido:\n")
print(round(modelo[["coefficients"]], 3), width = 56)
